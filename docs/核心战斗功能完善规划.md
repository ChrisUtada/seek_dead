### 模块1：体力与闪避系统

#### 1.1 体力消耗机制

**目标**：实现闪避/冲刺消耗体力，体力不足时限制行动

**实现要点**：

1. **体力消耗规则**
   - 闪避：消耗15体力
   - 冲刺：消耗10体力/秒
   - 体力恢复：10体力/秒（延迟1秒后开始恢复）
   - 体力不足阈值：< 5体力时无法闪避/冲刺

2. **体力状态管理**
   - 正常状态：体力充足，可正常行动
   - 体力不足状态：体力< 5，无法闪避/冲刺
   - 体力耗尽状态：体力=0，进入短暂眩晕

3. **UI反馈**
   - HUD体力条显示
   - 体力不足时体力条闪烁红色
   - 体力耗尽时显示提示文字

**技术实现**：

```gdscript
# 扩展 StateComponent
class_name StaminaSystem
extends Node

signal stamina_changed(current: float, max_value: float)
signal stamina_depleted()
signal stamina_low()

@export var max_stamina: float = 100.0
@export var recover_speed: float = 10.0
@export var recover_delay: float = 1.0

var current_stamina: float
var _recover_timer: float = 0.0
var _is_depleted: bool = false

func consume_stamina(amount: float) -> bool:
    if current_stamina < amount:
        return false
    current_stamina -= amount
    _recover_timer = recover_delay
    stamina_changed.emit(current_stamina, max_stamina)
    return true

func _process(delta: float):
    if _recover_timer > 0:
        _recover_timer -= delta
    else:
        if current_stamina < max_stamina:
            current_stamina += recover_speed * delta
            current_stamina = min(current_stamina, max_stamina)
            stamina_changed.emit(current_stamina, max_stamina)
```

**文件清单**：
- `scripts/battle/stamina_system.gd` - 体力系统核心
- `scripts/components/dodge_component.gd` - 闪避组件
- `scripts/components/sprint_component.gd` - 冲刺组件

   - 闪避时间：0.2秒
   - 闪避冷却：0.5秒
   - 体力消耗：15

2. **闪避行为**
   - 按下闪避键（Shift）触发
   - 向移动方向闪避，无移动时向后方闪避
   - 闪避期间短暂无敌（0.15秒）
   - 闪避结束后有短暂僵直（0.1秒）

3. **视觉效果**
   - 闪避时角色残影
   - 闪避时角色半透明
   - 闪避方向指示

**技术实现**：

```gdscript
class_name DodgeComponent
extends Node

signal dodge_started(direction: Vector2)
signal dodge_finished()

@export var dodge_distance: float = 150.0
@export var dodge_duration: float = 0.2
@export var dodge_cooldown: float = 0.5
@export var stamina_cost: float = 15.0

var _dodge_timer: float = 0.0
var _cooldown_timer: float = 0.0
var _is_dodging: bool = false

func can_dodge() -> bool:
    return _cooldown_timer <= 0.0 and not _is_dodging

func dodge(direction: Vector2):
    if not can_dodge():
        return
    
    var stamina = get_parent().get_node("StaminaSystem")
    if not stamina.consume_stamina(stamina_cost):
        return
    
    _is_dodging = true
    _dodge_timer = dodge_duration
    _cooldown_timer = dodge_cooldown
    
    # 执行闪避移动
    var tween = create_tween()
    var start_pos = get_parent().global_position
    var end_pos = start_pos + direction * dodge_distance
    
    tween.tween_property(get_parent(), "global_position", end_pos, dodge_duration)
    tween.tween_callback(_on_dodge_finished)
    
    dodge_started.emit(direction)

func _process(delta: float):
    if _dodge_timer > 0:
        _dodge_timer -= delta
    if _cooldown_timer > 0:
        _cooldown_timer -= delta

func _on_dodge_finished():
    _is_dodging = false
    dodge_finished.emit()
```

```gdscript
class_name SprintComponent
extends Node

signal sprint_started()
signal sprint_finished()

@export var speed_multiplier: float = 1.5
@export var stamina_cost_per_second: float = 10.0

var _is_sprinting: bool = false

func start_sprint():
    if _is_sprinting:
        return
    
    var stamina = get_parent().get_node("StaminaSystem")
    if stamina.current_stamina < stamina_cost_per_second:
        return
    
    _is_sprinting = true
    get_parent().get_node("MovementComponent").speed_multiplier = speed_multiplier
    sprint_started.emit()

func stop_sprint():
    if not _is_sprinting:
        return
    
    _is_sprinting = false
    get_parent().get_node("MovementComponent").reset_speed_multiplier()
    sprint_finished.emit()

func _process(delta: float):
    if _is_sprinting:
        var stamina = get_parent().get_node("StaminaSystem")
        if not stamina.consume_stamina(stamina_cost_per_second * delta):
            stop_sprint()
```

### 模块2：武器过热系统

#### 2.1 热量累积机制

**目标**：攻击累积热量，热量过高时武器性能下降

**实现要点**：

1. **热量参数**
   - 最大热量：100
   - 冷却速度：8热量/秒
   - 预热阈值：75热量（进入过热警告）
   - 过热阈值：100热量（强制冷却）

2. **热量累积**
   - 每次攻击增加热量（基于武器类型）
   - 近战武器：5热量/次
   - 远程武器：8热量/次
   - 攻击后延迟1秒开始冷却

3. **过热效果**
   - 预热状态（75-99热量）：攻击速度降低20%
   - 过热状态（100热量）：无法攻击，强制冷却5秒
   - 过热警告：热量条闪烁橙色

**技术实现**：

```gdscript
class_name HeatSystem
extends Node

signal heat_changed(current: float, max_value: float)
signal overheat_warning()
signal overheat_triggered()
signal overheat_cleared()

@export var max_heat: float = 100.0
@export var cooling_speed: float = 8.0
@export var warning_threshold: float = 75.0
@export var overheat_threshold: float = 100.0

var current_heat: float = 0.0
var _cooling_timer: float = 0.0
var _is_overheated: bool = false

func add_heat(amount: float):
    if _is_overheated:
        return
    
    current_heat += amount
    current_heat = min(current_heat, max_heat)
    _cooling_timer = 1.0  # 攻击后延迟冷却
    heat_changed.emit(current_heat, max_heat)
    
    if current_heat >= overheat_threshold:
        _trigger_overheat()
    elif current_heat >= warning_threshold:
        overheat_warning.emit()

func _trigger_overheat():
    _is_overheated = true
    current_heat = max_heat
    overheat_triggered.emit()
    
    # 强制冷却5秒
    var tween = create_tween()
    tween.tween_interval(5.0)
    tween.tween_callback(_clear_overheat)

func _clear_overheat():
    _is_overheated = false
    current_heat = 0.0
    overheat_cleared.emit()

func _process(delta: float):
    if _cooling_timer > 0:
        _cooling_timer -= delta
    else:
        if current_heat > 0 and not _is_overheated:
            current_heat -= cooling_speed * delta
            current_heat = max(current_heat, 0.0)
            heat_changed.emit(current_heat, max_heat)

func get_heat_ratio() -> float:
    return current_heat / max_heat if max_heat > 0 else 0.0

func is_overheated() -> bool:
    return _is_overheated

func is_warning() -> bool:
    return current_heat >= warning_threshold and current_heat < overheat_threshold
```

   - 商店购买弹药

**技术实现**：

```gdscript
class_name AmmoSystem
extends Node

signal ammo_changed(current: int, max_capacity: int)
signal reload_started()
signal reload_finished()
signal ammo_empty()

@export var max_ammo: int = 30
@export var reload_time: float = 2.0

var current_ammo: int
var _is_reloading: bool = false
var _reload_timer: float = 0.0

func _ready():
    current_ammo = max_ammo

func consume_ammo() -> bool:
    if _is_reloading:
        return false
    
    if current_ammo <= 0:
        ammo_empty.emit()
        start_reload()
        return false
    
    current_ammo -= 1
    ammo_changed.emit(current_ammo, max_ammo)
    
    if current_ammo <= 0:
        ammo_empty.emit()
        start_reload()
    
    return true

func start_reload():
    if _is_reloading:
        return
    
    _is_reloading = true
    _reload_timer = reload_time
    reload_started.emit()

func _process(delta: float):
    if _is_reloading:
        _reload_timer -= delta
        if _reload_timer <= 0:
            current_ammo = max_ammo
            _is_reloading = false
            ammo_changed.emit(current_ammo, max_ammo)
            reload_finished.emit()

func add_ammo(amount: int):
    current_ammo = min(current_ammo + amount, max_ammo)
    ammo_changed.emit(current_ammo, max_ammo)

func get_ammo_ratio() -> float:
    return float(current_ammo) / float(max_ammo) if max_ammo > 0 else 0.0
```

```gdscript
    skill_finished.emit()

func _process(delta: float):
```

```gdscript
class_name SkillManager
extends Node

signal skill_used(skill_index: int, skill: SkillBase)
signal skill_cooldown_changed(skill_index: int, ratio: float)

var skills: Array[SkillBase] = []
var _skill_cooldowns: Array[float] = []

func add_skill(skill: SkillBase):
    skills.append(skill)
    _skill_cooldowns.append(0.0)

func use_skill(index: int, user: Node2D) -> bool:
    if index < 0 or index >= skills.size():
        return false
    
    var skill = skills[index]
    var state = user.get_node("StateComponent")
    
    if not skill.can_use(state.energy):
        return false
    
    if skill.use(user):
        skill_used.emit(index, skill)
        return true
    
    return false

func _process(delta: float):
    for i in range(skills.size()):
        if _skill_cooldowns[i] > 0:
            _skill_cooldowns[i] -= delta
            skill_cooldown_changed.emit(i, _skill_cooldowns[i] / skills[i].cooldown)

func get_skill_cooldown_ratio(index: int) -> float:
    if index < 0 or index >= skills.size():
        return 0.0
    return _skill_cooldowns[index] / skills[index].cooldown if skills[index].cooldown > 0 else 0.0
```

## 五、技术难点与解决方案

### 5.1 体力系统与状态机集成

**难点**：体力耗尽需要与角色状态机集成，进入特殊状态

**解决方案**：
- 在 `StateComponent` 中添加体力耗尽状态
- 使用信号机制通知状态变化
- 状态机监听体力变化信号

### 5.2 热量系统与武器系统耦合

**难点**：热量需要影响武器性能，但又要保持系统解耦

**解决方案**：
- 使用组件化设计，热量系统和武器系统独立
- 通过信号和查询接口进行通信
- 武器系统查询热量状态来修正性能

### 5.3 技能系统的扩展性

**难点**：技能系统需要支持多种不同类型的技能

**解决方案**：
- 使用继承和多态，`SkillBase` 作为基类
- 每个具体技能继承 `SkillBase` 并实现 `_activate_skill`
- 使用资源文件配置技能参数

### 5.4 UI实时更新

**难点**：多个系统状态需要实时更新UI

**解决方案**：
- 使用信号机制，状态变化时发送信号
- UI监听信号并更新显示
- 使用缓动动画平滑过渡
